Task 5

Add & remove elements

A list with Add Item and Remove Item buttons that use .append(), .prepend() and .remove().

  • Item 1 — created in the markuporiginal
  • Item 2 — created in the markuporiginal

// Task 5
$("#btn-append").on("click", function () {
  $("#item-list").append("<li>Item " + counter + "</li>");   // add at the end
});

$("#btn-prepend").on("click", function () {
  $("#item-list").prepend("<li>Item " + counter + "</li>");  // add at the top
});

$("#btn-remove").on("click", function () {
  $("#item-list li").last().remove();                      // delete an element
});

Modifying attributes

// Task 6
$("#btn-change-src").on("click", function () {
  var next = $("#attr-image").attr("src") === "images/sunset.svg"
    ? "images/ocean.svg" : "images/sunset.svg";
  $("#attr-image").attr("src", next);          // rewrite the attribute
});

$("#btn-change-href").on("click", function () {
  $("#attr-link").attr("href", "https://www.w3schools.com/jquery/");
});
